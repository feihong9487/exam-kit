【情境 S01：位元欄位取出／填入／符號延伸】
重點：遮罩用 0xFFFFFFFFu>>(32-w)，w=32 也安全；不要寫 (1u<<w)-1（w=32 時是未定義行為）。w=0 要特判。
陷阱：填入時 v 要先 & 遮罩，否則會污染隔壁欄位；符號延伸用 (v^sign)-sign；全部用 uint32_t，不要用 int 做位移。
#include <stdio.h>
#include <stdint.h>
#define MASK(w) (0xFFFFFFFFu >> (32 - (w)))
#define GETF(x,hi,lo) (((uint32_t)(x) >> (lo)) & MASK((hi)-(lo)+1))
#define SETF(x,hi,lo,v) (((uint32_t)(x) & ~(MASK((hi)-(lo)+1) << (lo))) | (((uint32_t)(v) & MASK((hi)-(lo)+1)) << (lo)))
uint32_t get_bits(uint32_t x, int lo, int w) {          /* 從第 lo 位取 w 位 */
    if (w <= 0) return 0;
    return (x >> lo) & MASK(w);
}
uint32_t set_bits(uint32_t x, int lo, int w, uint32_t v) { /* 把 v 填進 [lo, lo+w) */
    if (w <= 0) return x;
    uint32_t m = MASK(w) << lo;
    return (x & ~m) | ((v << lo) & m);
}
int32_t sign_ext(uint32_t v, int w) {                   /* w 位元二補數 -> int32 */
    if (w <= 0) return 0;
    if (w >= 32) return (int32_t)v;
    uint32_t s = 1u << (w - 1);
    v &= MASK(w);
    return (int32_t)((v ^ s) - s);
}
int32_t get_sbits(uint32_t x, int lo, int w) { return sign_ext(get_bits(x, lo, w), w); }
int main(void) {
    printf("%X\n", get_bits(0xABCD1234u, 4, 8));   /* 23 */
    printf("%X\n", set_bits(0, 4, 4, 0xF));        /* F0 */
    printf("%d\n", sign_ext(0x800, 12));           /* -2048 */
    printf("%u\n", GETF(0x00500093u, 31, 20));     /* 5 */
    printf("%X\n", SETF(0u, 11, 7, 5));            /* 280 */
    printf("%d\n", get_sbits(0xfff00093u, 20, 12));/* -1 */
    return 0;
}

【情境 S02：依規格重排位元（反轉／nibble、byte 交換／旋轉／對照表搬移）】
重點：先看規格是「輸出 bit j 來自輸入 bit perm[j]」(gather) 還是「輸入 bit i 搬到 perm[i]」(scatter)，兩者互為反運算，用錯答案就整個反過來。
陷阱：uint8_t/uint16_t 運算會被提升成 int，結果要轉型回去；旋轉 r=0 時 x>>(32-r) 是未定義行為要特判；8/16 位反轉用 rev_n(x,8)/rev_n(x,16)。
#include <stdio.h>
#include <stdint.h>
uint32_t rev_n(uint32_t x, int n) {          /* 反轉低 n 位 */
    uint32_t r = 0;
    for (int i = 0; i < n; i++) r = (r << 1) | ((x >> i) & 1u);
    return r;
}
uint32_t rev32(uint32_t x) {
    x = ((x >> 1) & 0x55555555u) | ((x & 0x55555555u) << 1);
    x = ((x >> 2) & 0x33333333u) | ((x & 0x33333333u) << 2);
    x = ((x >> 4) & 0x0F0F0F0Fu) | ((x & 0x0F0F0F0Fu) << 4);
    x = ((x >> 8) & 0x00FF00FFu) | ((x & 0x00FF00FFu) << 8);
    return (x >> 16) | (x << 16);
}
uint32_t swap_nibbles32(uint32_t x) { return ((x & 0x0F0F0F0Fu) << 4) | ((x & 0xF0F0F0F0u) >> 4); }
uint32_t bswap32(uint32_t x) { return (x >> 24) | ((x >> 8) & 0xFF00u) | ((x << 8) & 0xFF0000u) | (x << 24); }
uint32_t rotl32(uint32_t x, unsigned r) { r &= 31; return r ? (x << r) | (x >> (32 - r)) : x; }
uint32_t gather(uint32_t x, const int *p, int n) {   /* 輸出 j = 輸入 p[j] */
    uint32_t r = 0;
    for (int j = 0; j < n; j++) r |= ((x >> p[j]) & 1u) << j;
    return r;
}
uint32_t scatter(uint32_t x, const int *p, int n) {  /* 輸入 i -> 輸出 p[i] */
    uint32_t r = 0;
    for (int i = 0; i < n; i++) r |= ((x >> i) & 1u) << p[i];
    return r;
}
int main(void) {
    int perm[8] = {1, 0, 3, 2, 5, 4, 7, 6};
    printf("%02X\n", rev_n(0xB4, 8));              /* 2D */
    printf("%04X\n", rev_n(0x1234, 16));           /* 2C48 */
    printf("%08X\n", rev32(0x12345678u));          /* 1E6A2C48 */
    printf("%08X\n", swap_nibbles32(0x12345678u)); /* 21436587 */
    printf("%08X\n", bswap32(0x12345678u));        /* 78563412 */
    printf("%08X\n", rotl32(0x12345678u, 8));      /* 34567812 */
    printf("%02X\n", gather(0xB4, perm, 8));       /* 78 */
    printf("%02X\n", scatter(0x01, perm, 8));      /* 02 */
    return 0;
}

【情境 S03：位元流讀取器（MSB／LSB-first 讀 n 位元欄位）】
重點：MSB-first＝每個 byte 從 bit7 讀起、先讀到的位元放欄位高位；LSB-first＝從 bit0 讀起、先讀到的放低位（等於把 bytes 當小端整數）。
陷阱：有些格式是混合規則（例如位元 MSB-first 取、值卻用 LSB 組），一定要拿題目範例手算一次；讀超過尾端要報錯；n=32 時不要用 1u<<n。
#include <stdio.h>
#include <stdint.h>
#include <stddef.h>
typedef struct { const uint8_t *buf; size_t nbits, pos; int lsb, err; } BR;
void br_init(BR *b, const uint8_t *buf, size_t nbytes, int lsb) {
    b->buf = buf; b->nbits = nbytes * 8; b->pos = 0; b->lsb = lsb; b->err = 0;
}
uint32_t br_read(BR *b, int n) {               /* n = 1..32 */
    uint32_t v = 0;
    if (n <= 0 || n > 32 || b->pos + (size_t)n > b->nbits) { b->err = 1; return 0; }
    for (int i = 0; i < n; i++) {
        uint8_t by = b->buf[b->pos >> 3];
        int k = (int)(b->pos & 7);
        b->pos++;
        uint32_t bit = b->lsb ? (by >> k) & 1u : (by >> (7 - k)) & 1u;
        if (b->lsb) v |= bit << i;             /* 先讀到的放低位 */
        else        v = (v << 1) | bit;        /* 先讀到的放高位 */
    }
    return v;
}
int main(void) {
    const uint8_t a[] = {0xB4, 0x5F};          /* 1011 0100 0101 1111 */
    int w[] = {3, 5, 4, 4};
    BR b;
    br_init(&b, a, 2, 0);
    for (int i = 0; i < 4; i++) printf("%u ", br_read(&b, w[i]));
    printf("\n");                              /* 5 20 5 15 */
    br_init(&b, a, 2, 1);
    for (int i = 0; i < 4; i++) printf("%u ", br_read(&b, w[i]));
    printf("\n");                              /* 4 22 15 5 */
    const uint8_t c[] = {0x12, 0x34, 0x56};
    br_init(&b, c, 3, 0);
    printf("%X\n", br_read(&b, 12));           /* 123 */
    br_init(&b, c, 3, 1);
    printf("%X\n", br_read(&b, 12));           /* 412 */
    return 0;
}

【情境 S04：位元流寫入器（把 n 位元欄位依序打包成 byte 陣列）】
重點：MSB-first 先寫欄位最高位、放在 byte 的 bit7 往下；LSB-first 先寫最低位、放在 bit0 往上。用 |= 寫入，所以緩衝區一定要先 memset 成 0。
陷阱：總位元數不是 8 的倍數時最後一個 byte 補 0，輸出 byte 數 = (pos+7)/8；v 超過 n 位的部分不能寫進去；讀回驗證請配 S03 的讀取器（同一種模式）。
#include <stdio.h>
#include <stdint.h>
#include <string.h>
typedef struct { uint8_t *buf; size_t cap, pos; int lsb; } BW;
void bw_init(BW *w, uint8_t *buf, size_t nbytes, int lsb) {
    memset(buf, 0, nbytes);
    w->buf = buf; w->cap = nbytes * 8; w->pos = 0; w->lsb = lsb;
}
int bw_write(BW *w, uint32_t v, int n) {       /* n = 1..32，空間不夠回傳 -1 */
    if (n <= 0 || n > 32 || w->pos + (size_t)n > w->cap) return -1;
    for (int i = 0; i < n; i++) {
        uint32_t b = w->lsb ? (v >> i) & 1u : (v >> (n - 1 - i)) & 1u;
        size_t by = w->pos >> 3;
        int k = (int)(w->pos & 7);
        if (b) w->buf[by] |= (uint8_t)(w->lsb ? (1u << k) : (0x80u >> k));
        w->pos++;
    }
    return 0;
}
size_t bw_bytes(const BW *w) { return (w->pos + 7) / 8; }
void dump(const uint8_t *b, size_t n) {
    for (size_t i = 0; i < n; i++) printf("%02X ", b[i]);
    printf("\n");
}
int main(void) {
    uint8_t buf[16];
    BW w;
    bw_init(&w, buf, sizeof buf, 0);           /* MSB-first */
    bw_write(&w, 5, 3); bw_write(&w, 20, 5); bw_write(&w, 5, 4); bw_write(&w, 15, 4);
    dump(buf, bw_bytes(&w));                   /* B4 5F */
    bw_init(&w, buf, sizeof buf, 1);           /* LSB-first */
    bw_write(&w, 4, 3); bw_write(&w, 22, 5); bw_write(&w, 15, 4); bw_write(&w, 5, 4);
    dump(buf, bw_bytes(&w));                   /* B4 5F */
    bw_init(&w, buf, sizeof buf, 0);
    bw_write(&w, 1, 1);
    dump(buf, bw_bytes(&w));                   /* 80 */
    return 0;
}

【情境 S05：校驗碼 CRC-32／CRC-16-CCITT／加總／同位元】
重點：CRC-32(IEEE)＝反射版 poly 0xEDB88320、init 與 xorout 0xFFFFFFFF，"123456789"→CBF43926；CRC-16-CCITT＝poly 0x1021、init 0xFFFF、不反射→29B1。
陷阱：題目給新參數就用 crc_generic(寬度,poly,init,refin,refout,xorout)，poly 填「不反射」寫法（例 0x04C11DB7）。RP2350 sniffer 0x0/0x2/0xf/0xe＝CRC32/CRC16/加總/XOR 化簡，位元順序是否同標準未查證，以題目為準。
#include <stdio.h>
#include <stdint.h>
#include <stddef.h>
uint32_t reflect(uint32_t v, int n) { uint32_t r = 0; for (int i = 0; i < n; i++) r = (r << 1) | ((v >> i) & 1u); return r; }
uint32_t crc_generic(const uint8_t *d, size_t len, int w, uint32_t poly,
                     uint32_t init, int refin, int refout, uint32_t xorout) {
    uint32_t mask = 0xFFFFFFFFu >> (32 - w), top = 1u << (w - 1), c = init & mask;
    for (size_t i = 0; i < len; i++) {
        uint32_t b = refin ? reflect(d[i], 8) : d[i];
        for (int k = 7; k >= 0; k--) {
            uint32_t in = (b >> k) & 1u, msb = (c & top) ? 1u : 0u;
            c = (c << 1) & mask;
            if (msb ^ in) c ^= poly;
        }
    }
    if (refout) c = reflect(c, w);
    return (c ^ xorout) & mask;
}
uint32_t crc32_ieee(const uint8_t *d, size_t len) {
    uint32_t c = 0xFFFFFFFFu;
    for (size_t i = 0; i < len; i++) {
        c ^= d[i];
        for (int k = 0; k < 8; k++) c = (c & 1u) ? (c >> 1) ^ 0xEDB88320u : c >> 1;
    }
    return ~c;
}
uint32_t crc16_ccitt(const uint8_t *d, size_t len) {
    uint32_t c = 0xFFFF;
    for (size_t i = 0; i < len; i++) {
        c ^= (uint32_t)d[i] << 8;
        for (int k = 0; k < 8; k++) c = (c & 0x8000u) ? ((c << 1) ^ 0x1021u) : (c << 1);
        c &= 0xFFFFu;
    }
    return c;
}
uint32_t sum32(const uint8_t *d, size_t len) { uint32_t s = 0; for (size_t i = 0; i < len; i++) s += d[i]; return s; }
uint32_t parity(const uint8_t *d, size_t len) {
    uint32_t x = 0;
    for (size_t i = 0; i < len; i++) x ^= d[i];
    x ^= x >> 4; x ^= x >> 2; x ^= x >> 1;
    return x & 1u;
}
int main(void) {
    const uint8_t *m = (const uint8_t *)"123456789";
    printf("%08X\n", crc32_ieee(m, 9));   /* CBF43926 */
    printf("%04X\n", crc16_ccitt(m, 9));  /* 29B1 */
    printf("%08X\n", crc_generic(m, 9, 32, 0x04C11DB7u, 0xFFFFFFFFu, 1, 1, 0xFFFFFFFFu)); /* CBF43926 */
    printf("%X %u\n", sum32(m, 9), parity(m, 9)); /* 1DD 1 */
    return 0;
}

【情境 S07：RV32I 單條組譯器（欄位→32 位元機器碼）】
重點：每種格式一個 enc 函式，欄位由高到低：R=f7|rs2|rs1|f3|rd|op，I=imm[11:0]|rs1|f3|rd|op，U=imm20|rd|op。opcode：OP 0x33、OP-IMM 0x13、LOAD 0x03、STORE 0x23、BRANCH 0x63、LUI 0x37、JAL 0x6F、JALR 0x67；sub/srai 的 f7=0x20，M 擴充 f7=1。
陷阱：S 的 imm 拆 [11:5]/[4:0]；B 是 13 位 [12|10:5|4:1|11]、J 是 21 位 [20|10:1|11|19:12]，位移必須偶數；srai 的 imm 要填 0x400|shamt；sw 的 rs2 是要存的值；I 型 imm 範圍 -2048..2047。
#include <stdio.h>
#include <stdint.h>
uint32_t enc_r(uint32_t f7, uint32_t rs2, uint32_t rs1, uint32_t f3, uint32_t rd, uint32_t op) {
    return f7 << 25 | rs2 << 20 | rs1 << 15 | f3 << 12 | rd << 7 | op;
}
uint32_t enc_i(int32_t imm, uint32_t rs1, uint32_t f3, uint32_t rd, uint32_t op) {
    return ((uint32_t)imm & 0xFFF) << 20 | rs1 << 15 | f3 << 12 | rd << 7 | op;
}
uint32_t enc_s(int32_t imm, uint32_t rs2, uint32_t rs1, uint32_t f3, uint32_t op) {
    uint32_t u = (uint32_t)imm;
    return ((u >> 5) & 0x7F) << 25 | rs2 << 20 | rs1 << 15 | f3 << 12 | (u & 31) << 7 | op;
}
uint32_t enc_b(int32_t imm, uint32_t rs2, uint32_t rs1, uint32_t f3, uint32_t op) {
    uint32_t u = (uint32_t)imm;
    return ((u >> 12) & 1) << 31 | ((u >> 5) & 0x3F) << 25 | rs2 << 20 | rs1 << 15 | f3 << 12
         | ((u >> 1) & 0xF) << 8 | ((u >> 11) & 1) << 7 | op;
}
uint32_t enc_u(uint32_t imm20, uint32_t rd, uint32_t op) { return (imm20 & 0xFFFFF) << 12 | rd << 7 | op; }
uint32_t enc_j(int32_t imm, uint32_t rd, uint32_t op) {
    uint32_t u = (uint32_t)imm;
    return ((u >> 20) & 1) << 31 | ((u >> 1) & 0x3FF) << 21 | ((u >> 11) & 1) << 20
         | ((u >> 12) & 0xFF) << 12 | rd << 7 | op;
}
int main(void) {
    printf("%08x\n", enc_i(5, 0, 0, 1, 0x13));          /* addi x1,x0,5    00500093 */
    printf("%08x\n", enc_s(8, 5, 2, 2, 0x23));          /* sw x5,8(x2)     00512423 */
    printf("%08x\n", enc_b(-4, 0, 0, 0, 0x63));         /* beq x0,x0,-4    fe000ee3 */
    printf("%08x\n", enc_r(0x20, 11, 10, 0, 10, 0x33)); /* sub x10,x10,x11 40b50533 */
    printf("%08x\n", enc_u(0x12345, 1, 0x37));          /* lui x1,0x12345  123450b7 */
    printf("%08x\n", enc_j(-4, 0, 0x6F));               /* jal x0,-4       ffdff06f */
    printf("%08x\n", enc_r(1, 2, 1, 0, 3, 0x33));       /* mul x3,x1,x2    022081b3 */
    return 0;
}

【情境 S08：迷你 RV32I 模擬器（32 暫存器＋記憶體，跑指令陣列）】
重點：迴圈 取指(pc)→解碼→執行→pc=next；x0 永遠 0（寫入前檢查 rd!=0）；分支/jal 目標＝pc+imm（不是 pc+4+imm）；blt 要轉 int32_t 比較。
陷阱：jalr 先算目標再寫 rd（rd 可能等於 rs1），目標 &~1；記憶體小端；要有步數上限與非法指令報錯。本精簡版記憶體以 word 存，lw/sw 假設 4 對齊。
#include <stdio.h>
#include <stdint.h>
#include <string.h>
#define F(x,h,l) (((x)>>(l))&(0xFFFFFFFFu>>(31-((h)-(l)))))
int32_t sx(uint32_t v,int w){uint32_t s=1u<<(w-1);v&=(s<<1)-1u;return (int32_t)((v^s)-s);}
uint32_t X[32],mem[1024];                 /* 4 KB */
int run(const uint32_t*p,int n,int max){  /* 回傳步數，錯誤 -1 */
 memset(X,0,sizeof X);memset(mem,0,sizeof mem);memcpy(mem,p,(size_t)n*4);
 uint32_t pc=0;

【情境 S10：PIO 狀態機 PC 模擬器（C）】
規則：OSR 計數重設=32(空)、ISR=0；autopull：OUT 前計數≥門檻就從 TX 補，TX 空就 stall；autopush：IN 後計數≥門檻就推 RX，RX 滿 stall；門檻 0 代表 32。

API（engine.h，只有 4 通道）：int engine_claim(void) 全滿回 -1；void engine_configure(int ch,const engine_job_t*)；engine_start/engine_busy/engine_wait/engine_release(int ch)。

#include <stdint.h>
#include <stdio.h>

【情境 S17：有限資源池（semaphore 初值 4 管 4 個通道，10 個工作排隊）】
- sem_wait：沒空位就睡著排隊（第五個請求會等到有人 release）；sem_trywait：拿不到回 -1、errno==EAGAIN，可先做別的事再試。
- semaphore 只管「還剩幾個」，「哪一個通道空著」要另外用 mutex 保護的 in_use[] 找。
- 歸還順序：先把 in_use[ch]=0，再 sem_post；每條路徑（含錯誤）都要 release，否則通道永遠少一個。
- 實測：10 個工作同時最多 4 個在用（max_busy=4），結束後 sem 值回到 4。

#include <errno.h>
#include <pthread.h>
#include <semaphore.h>
#include <stdio.h>
#include <unistd.h>
#define CHANNELS 4
#define JOBS 10
static sem_t free_ch;
static pthread_mutex_t mtx = PTHREAD_MUTEX_INITIALIZER;
static int in_use[CHANNELS];
static int claim(void) {                 /* 呼叫前已拿到 sem，一定找得到 */
    pthread_mutex_lock(&mtx);
    int ch = -1;
    for (int i = 0; i < CHANNELS; i++) if (!in_use[i]) { in_use[i] = 1; ch = i; break; }
    pthread_mutex_unlock(&mtx);
    return ch;
}
static void release(int ch) {
    pthread_mutex_lock(&mtx); in_use[ch] = 0; pthread_mutex_unlock(&mtx);
    sem_post(&free_ch);
}
static void *job_blocking(void *p) {     /* 版本 A：排隊等 */
    int id = (int)(long)p;
    sem_wait(&free_ch);
    int ch = claim();
    printf("job %d got ch %d\n", id, ch);
    usleep(20000);                       /* 使用通道 */
    release(ch);
    return NULL;
}
static void *job_try(void *p) {          /* 版本 B：拿不到先做別的 */
    int id = (int)(long)p; long other = 0;
    while (sem_trywait(&free_ch) != 0) {
        if (errno != EAGAIN) return NULL;
        other++; usleep(1000);           /* 做別的事 */
    }
    int ch = claim();
    printf("job %d got ch %d after %ld other tasks\n", id, ch, other);
    usleep(20000);
    release(ch);
    return NULL;
}
int main(void) {
    pthread_t th[JOBS];
    sem_init(&free_ch, 0, CHANNELS);     /* 0=執行緒間共用，初值=通道數 */
    for (long i = 0; i < JOBS; i++) pthread_create(&th[i], NULL, job_blocking, (void *)i);
    for (int i = 0; i < JOBS; i++) pthread_join(th[i], NULL);
    for (long i = 0; i < JOBS; i++) pthread_create(&th[i], NULL, job_try, (void *)i);
    for (int i = 0; i < JOBS; i++) pthread_join(th[i], NULL);
    int v; sem_getvalue(&free_ch, &v);
    puts(v == CHANNELS ? "ALL OK" : "FAIL");
    sem_destroy(&free_ch);
    return v != CHANNELS;
}

【情境 S18：生產者－消費者（環形佇列＋mutex＋condition variable）】
- 等待一定寫 while (條件不成立) pthread_cond_wait(&cv, &m);（會被假喚醒、或被別人搶先拿走）。
- 放：滿了等 not_full，放完 signal not_empty；取：空了等 not_empty，取完 signal not_full。
- 結束訊號：生產者全部 join 後設 done=1 並 broadcast not_empty；消費者在「done 且空」時回 0 下班。漏了就永遠卡住。
- 實測：2 生產者×10 萬筆、3 消費者、容量 8，產出與消費的筆數與總和完全相同。

#include <pthread.h>
#include <stdio.h>
#define CAP 8
#define NP 2
#define NC 3
#define PER 100000
static int buf[CAP], head, tail, count, done;
static pthread_mutex_t m = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t not_empty = PTHREAD_COND_INITIALIZER, not_full = PTHREAD_COND_INITIALIZER;
static void put(int v) {
    pthread_mutex_lock(&m);
    while (count == CAP) pthread_cond_wait(&not_full, &m);
    buf[tail] = v; tail = (tail + 1) % CAP; count++;
    pthread_cond_signal(&not_empty);
    pthread_mutex_unlock(&m);
}
static int get(int *v) {                      /* 1=拿到，0=結束 */
    pthread_mutex_lock(&m);
    while (count == 0 && !done) pthread_cond_wait(&not_empty, &m);
    if (count == 0) { pthread_mutex_unlock(&m); return 0; }
    *v = buf[head]; head = (head + 1) % CAP; count--;
    pthread_cond_signal(&not_full);
    pthread_mutex_unlock(&m);
    return 1;
}
static long psum[NP], csum[NC];
static void *producer(void *p) {
    long id = (long)p;
    for (int i = 1; i <= PER; i++) { int v = (int)id * PER + i; put(v); psum[id] += v; }
    return NULL;
}
static void *consumer(void *p) {
    long id = (long)p; int v;
    while (get(&v)) csum[id] += v;
    return NULL;
}
int main(void) {
    pthread_t pt[NP], ct[NC];
    for (long i = 0; i < NC; i++) pthread_create(&ct[i], NULL, consumer, (void *)i);
    for (long i = 0; i < NP; i++) pthread_create(&pt[i], NULL, producer, (void *)i);
    for (int i = 0; i < NP; i++) pthread_join(pt[i], NULL);
    pthread_mutex_lock(&m); done = 1; pthread_cond_broadcast(&not_empty); pthread_mutex_unlock(&m);
    for (int i = 0; i < NC; i++) pthread_join(ct[i], NULL);
    long a = 0, b = 0;
    for (int i = 0; i < NP; i++) a += psum[i];
    for (int i = 0; i < NC; i++) b += csum[i];
    printf("%ld %ld\n%s\n", a, b, a == b ? "ALL OK" : "FAIL");
    return a != b;
}

【情境 S25：service server／client（std_srvs SetBool、Trigger）】
重點：ros-core 只有 std_srvs（Empty／SetBool／Trigger），沒有 example_interfaces 的 AddTwoInts。SetBool 請求 data(bool)，回應 success(bool)+message(str)；Trigger 請求沒欄位。
陷阱1：server callback 一定要 return response，忘了 → server 噴 TypeError 當掉、client 等不到。
陷阱2：client 用 call_async＋spin_until_future_complete（不要在 callback 裡用同步 call，會卡死）；先 wait_for_service。
陷阱3：Ctrl-C 要 except (KeyboardInterrupt, ExternalShutdownException)＋rclpy.try_shutdown()；service 名稱用相對名稱，__ns 不同就找不到。

import sys
import rclpy
from rclpy.node import Node
from rclpy.executors import ExternalShutdownException
from std_srvs.srv import SetBool, Trigger

class Server(Node):
    def __init__(self):
        super().__init__('robot_server')
        self.enabled = False
        self.create_service(SetBool, 'set_enable', self.on_set)
        self.create_service(Trigger, 'get_status', self.on_get)
    def on_set(self, req, res):
        self.enabled = req.data
        res.success = True; res.message = f'enabled={self.enabled}'
        return res
    def on_get(self, req, res):
        res.success = self.enabled; res.message = f'enabled={self.enabled}'
        return res

class Client(Node):
    def __init__(self):
        super().__init__('robot_client')
        self.cli = self.create_client(SetBool, 'set_enable')
    def call(self, value):
        if not self.cli.wait_for_service(timeout_sec=3.0):
            return None
        req = SetBool.Request(); req.data = value
        fut = self.cli.call_async(req)
        rclpy.spin_until_future_complete(self, fut, timeout_sec=3.0)
        return fut.result() if fut.done() else None

def main():
    argv = rclpy.utilities.remove_ros_args(sys.argv)
    role = argv[1] if len(argv) > 1 else 'server'
    rclpy.init(args=sys.argv)
    node = Server() if role == 'server' else Client()
    try:
        if role == 'server':
            rclpy.spin(node)
        else:
            print(node.call(len(argv) < 3 or argv[2] == 'true'))
    except (KeyboardInterrupt, ExternalShutdownException):
        pass
    finally:
        node.destroy_node()
        rclpy.try_shutdown()

if __name__ == '__main__':
    main()

import rclpy
from rclpy.node import Node
from rclpy.executors import ExternalShutdownException

from geometry_msgs.msg import Twist

class Robot(Node):
    def __init__(self):
        super().__init__('robot')
        self.declare_parameter('speed', 1.0)
        self.declare_parameter('robot_name', 'r')
        self.add_on_set_parameters_callback(self.on_param)
        self.cmd_pub = self.create_publisher(Twist, 'cmd_vel', 10)
        self.status_pub = self.create_publisher(String, 'status', 10)
        self.x = 0.0
        self.timer = self.create_timer(1.0, self.on_timer)

    def on_param(self, params):
        for p in params:
            if p.name == 'speed' and p.value < 0:
                return SetParametersResult(successful=False, reason='speed 不能是負的')
        return SetParametersResult(successful=True)

    def on_timer(self):
        speed = self.get_parameter('speed').value
        self.x += speed
        cmd = Twist(); cmd.linear.x = speed
        self.cmd_pub.publish(cmd)
        msg = String()
        msg.data = f'{self.get_parameter("robot_name").value} speed={speed} x={self.x:.1f}'
        self.status_pub.publish(msg)

def main():
    rclpy.init()
    node = Robot()
    try:
        rclpy.spin(node)
    except (KeyboardInterrupt, ExternalShutdownException):
        pass
    finally:
        node.destroy_node()
        rclpy.try_shutdown()

if __name__ == '__main__':

【情境 S27：一鍵排查腳本 ros2_check.sh（收不到資料時）】
用法：bash ros2_check.sh /double 8 → 印 node list、topic list -t、每個 topic pub/sub 數＋警告，再對 /double 做 info -v、echo --once、hz。

#!/usr/bin/env bash
TOPIC="${1:-}"; SEC="${2:-8}"
export PYTHONUNBUFFERED=1
[ -z "$ROS_DISTRO" ] && source /opt/ros/jazzy/setup.bash
ros2 topic list >/dev/null 2>&1; sleep 1
echo "== node list"; ros2 node list
echo "== topic list -t"; ros2 topic list -t
echo "== pub/sub"
TOPICS=$(ros2 topic list | grep -v -E '^/(rosout|parameter_events)$')
for t in $TOPICS; do
  ros2 topic info -v "$t" | awk -v T="$t" '
    /^Publisher count/ {np=$3}  /^Subscription count/ {ns=$3}
    /^Node name:/ {node=$3}     /^Endpoint type/ {ep=$3}
    /Reliability:/ {if (ep=="PUBLISHER") pr[$2]=pr[$2]" "node; else sr[$2]=sr[$2]" "node}
    END { printf "%-28s pub=%s sub=%s\n", T, np, ns
      if (np==0 && ns>0) print "  ! 有人收沒人發：名稱打錯／namespace？"
      if (np>=2) print "  ! " np " 個 publisher"
      if (("BEST_EFFORT" in pr) && ("RELIABLE" in sr)) print "  ! QoS 不相容" }'
done
if [ -n "$TOPIC" ]; then

【情境 S28：多機器人派單 dispatcher】
介面：機器人 -r __ns:=/robotK 啟動，內部用相對名稱 status(String "IDLE x y"／"BUSY T3 x y")、goal(PoseStamped，frame_id=任務編號)、done(String 任務編號)。
陷阱1：callback 別 sleep，用 timer 狀態機；迴圈 lambda 要寫 lambda m, n=n:，否則全綁最後一台。
陷阱2：dispatcher 無 namespace，用絕對名 f'/{n}/goal'；Ctrl-C：except (KeyboardInterrupt, ExternalShutdownException)＋rclpy.try_shutdown()。

import math
from collections import deque
import rclpy
from rclpy.node import Node
from rclpy.executors import ExternalShutdownException
from geometry_msgs.msg import PoseStamped
from std_msgs.msg import String

class Dispatcher(Node):
    def __init__(self):
        super().__init__('dispatcher')
        self.pending = deque([('T1', 2.0, 0.0), ('T2', 0.0, 3.0), ('T3', 5.0, 5.0)])
        self.total, self.done, self.finished, self.R = len(self.pending), 0, False, {}
        for n in ['robot1', 'robot2', 'robot3']:
            self.R[n] = {'online': False, 'pos': (0.0, 0.0), 'task': None,
                         'pub': self.create_publisher(PoseStamped, f'/{n}/goal', 10)}
            self.create_subscription(String, f'/{n}/status', lambda m, n=n: self.on_status(n, m), 10)
            self.create_subscription(String, f'/{n}/done', lambda m, n=n: self.on_done(n, m), 10)
        self.create_timer(0.2, self.tick)

    def on_status(self, n, m):
        p = m.data.split()
        self.R[n].update(online=True, pos=(float(p[-2]), float(p[-1])))

    def on_done(self, n, m):
        if self.R[n]['task'] == m.data:
            self.R[n]['task'] = None
            self.done += 1

    def tick(self):
        while self.pending:
            idle = [n for n, r in self.R.items() if r['online'] and r['task'] is None]
            if not idle:
                break
            t = self.pending.popleft()
            n = min(idle, key=lambda k: math.dist(self.R[k]['pos'], t[1:]))
            g = PoseStamped()
            g.header.frame_id = t[0]
            g.pose.position.x, g.pose.position.y = t[1], t[2]
            self.R[n]['pub'].publish(g)
            self.R[n]['task'] = t[0]
        self.finished = self.done == self.total

def main():
    rclpy.init()
    node = Dispatcher()
    try:
        while rclpy.ok() and not node.finished:
            rclpy.spin_once(node, timeout_sec=0.1)
    except (KeyboardInterrupt, ExternalShutdownException):
        pass
    finally:
        node.destroy_node()
        rclpy.try_shutdown()

if __name__ == '__main__':
    main()

【情境 S29：同一 node 的 pub＋sub＋timer＋統計（筆數、Hz、遺失）＋QoS】
重點：用遞增序號 Int32 判斷遺失：收到 s > last+1 → 遺失 s-last-1 筆；s <= last → 亂序／重複。每 report 秒印統計。
QoS：發 BEST_EFFORT＋收 RELIABLE 不相容（0 筆，rclpy 會 WARN incompatible QoS）；發 RELIABLE＋收 BEST_EFFORT 可以；TRANSIENT_LOCAL 兩端都設，晚到的訂閱者會先拿到最後 depth 筆。
陷阱：topic 用相對名稱（__ns 才分得開）；Ctrl-C 要 except (KeyboardInterrupt, ExternalShutdownException)＋rclpy.try_shutdown()。

import rclpy
from rclpy.node import Node
from rclpy.executors import ExternalShutdownException
from rclpy.qos import QoSProfile, ReliabilityPolicy, DurabilityPolicy, HistoryPolicy
from std_msgs.msg import Int32

class Stats(Node):
    def __init__(self):
        super().__init__('stats')
        qos = QoSProfile(history=HistoryPolicy.KEEP_LAST, depth=10,
                         reliability=ReliabilityPolicy.RELIABLE,      # 或 BEST_EFFORT
                         durability=DurabilityPolicy.VOLATILE)        # 或 TRANSIENT_LOCAL
        self.pub = self.create_publisher(Int32, 'seq', qos)
        self.create_subscription(Int32, 'seq', self.on_msg, qos)
        self.create_timer(0.1, self.on_pub)      # 10 Hz
        self.create_timer(2.0, self.on_report)   # 每 2 秒統計
        self.seq = 0
        self.total = self.lost = self.ooo = self.win = 0
        self.first = self.last = None
        self.t0 = self.now()

    def now(self):
        return self.get_clock().now().nanoseconds / 1e9

    def on_pub(self):
        self.seq += 1
        msg = Int32(); msg.data = self.seq
        self.pub.publish(msg)

    def on_msg(self, msg):
        s = msg.data
        self.total += 1; self.win += 1
        if self.first is None:
            self.first = self.last = s
        elif s > self.last:
            self.lost += s - self.last - 1
            self.last = s
        else:
            self.ooo += 1

    def on_report(self):
        t = self.now()
        hz = self.win / (t - self.t0)
        self.get_logger().info(f'收到 {self.total}｜{hz:.1f} Hz｜序號 {self.first}~{self.last}'
                               f'｜遺失 {self.lost}｜亂序 {self.ooo}')
        self.win, self.t0 = 0, t

def main():
    rclpy.init()
    node = Stats()
    try:
        rclpy.spin(node)
    except (KeyboardInterrupt, ExternalShutdownException):
        pass
    finally:
        node.destroy_node()
        rclpy.try_shutdown()

if __name__ == '__main__':
    main()

import sys
import rclpy
from rclpy.node import Node
from rclpy.action import ActionClient
from rclpy.executors import ExternalShutdownException
from action_msgs.msg import GoalStatus
from nav2_msgs.action import NavigateToPose

class NavClient(Node):
    def __init__(self):
        super().__init__('nav_client')
        self.client = ActionClient(self, NavigateToPose, 'navigate_to_pose')

    def go(self, x, y):
        if not self.client.wait_for_server(timeout_sec=5.0):
            self.get_logger().error('找不到 action server')
            return False
        goal = NavigateToPose.Goal()
        goal.pose.header.frame_id = 'map'
        goal.pose.header.stamp = self.get_clock().now().to_msg()
        goal.pose.pose.position.x, goal.pose.pose.position.y = x, y
        goal.pose.pose.orientation.w = 1.0
        f = self.client.send_goal_async(goal, feedback_callback=self.on_fb)
        rclpy.spin_until_future_complete(self, f)
        handle = f.result()
        if not handle.accepted:
            return False
        rf = handle.get_result_async()
        rclpy.spin_until_future_complete(self, rf, timeout_sec=60.0)
        if not rf.done():
            handle.cancel_goal_async()
            return False
        return rf.result().status == GoalStatus.STATUS_SUCCEEDED

    def on_fb(self, msg):
        self.get_logger().info(f'剩 {msg.feedback.distance_remaining:.2f} m')

def main():
    argv = rclpy.utilities.remove_ros_args(sys.argv)
    rclpy.init(args=sys.argv)
    node = NavClient()
    try:
        ok = node.go(float(argv[1]), float(argv[2]))
        print('成功' if ok else '失敗')
    except (KeyboardInterrupt, ExternalShutdownException):
        pass
    finally:
        node.destroy_node()
        rclpy.try_shutdown()

if __name__ == '__main__':
    main()
